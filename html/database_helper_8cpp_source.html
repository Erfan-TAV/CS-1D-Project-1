81void calculateEfficientTrip(int currentID, QList<int> unvisitedIDs, double totalDistance, int order) {
82 // BASE CASE: If there are no more campuses to visit, stop the recursion.
83 if (unvisitedIDs.isEmpty()) {
84 qDebug() << "[ALGO] --- Recursion Complete ---";
85 qDebug() << "[ALGO] Final Total Distance:" << totalDistance << "miles.";
86 return;
87 }
88 int nextID = -1;
89 double minFound = std::numeric_limits<double>::max();
90 // Find the single closest school from the current location
91 for (int targetID : unvisitedIDs) {
92 double dist = getDistanceBetween(currentID, targetID);
93 if (dist < minFound) {
94 minFound = dist;
95 nextID = targetID;
96 }
97 }
98 if (nextID != -1) {
99 // Prepare data for the next stop
100 double newTotalDistance = totalDistance + minFound;
101 QString name = getCampusName(nextID);
102 qDebug() << "[ALGO] Recursive Level" << order << ": Visiting" << name
103 << " (ID:" << nextID << ") +" << minFound << "mi.";
104 // Record the visit in the database
105 addTripCampus(nextID, name, order);
106 // Remove the campus we just visited
107 unvisitedIDs.removeAll(nextID);
108 // RECURSIVE CALL: The function calls ITSELF with the new location and updated totals
109 calculateEfficientTrip(nextID, unvisitedIDs, newTotalDistance, order + 1);
110 } else {
111 qDebug() << "[ALGO] ERROR: Path broken at ID" << currentID;
112 }

171int closestCampus(const int campusID) {
172 QSqlQuery query;
173 int minDistance = INT_MAX;
174 int nearestID = -1;
175 // The SQL logic:
176 // 1. Check if our ID is in column 1 OR column 2.
177 // 2. Use CASE to return the "other" ID as 'neighborID'.
178 query.prepare("SELECT "
179 "CASE WHEN campusID1 = :id THEN campusID2 ELSE campusID1 END AS neighborID, "
180 "distance "
181 "FROM campusDistances "
182 "WHERE campusID1 = :id OR campusID2 = :id");
183 query.bindValue(":id", campusID);
184 if (!query.exec()) {
185 qDebug() << "SQL ERROR:" << query.lastError().text();
186 return -1;
187 }
188 while (query.next()) {
189 int neighborID = query.value(0).toInt();
190 int currentDist = query.value(1).toInt();
191 // Standard greedy check for the minimum distance
192 if (currentDist < minDistance) {
193 minDistance = currentDist;
194 nearestID = neighborID;
195 }
196 }
197 if (nearestID == -1) {
198 qDebug() << "No connections found for Campus ID:" << campusID;
199 }
200 return nearestID;

288void resetAndReloadData(const QString &filePath) {
289 QXlsx::Document xlsx(filePath);
290 if (!xlsx.load()) {
291 qDebug() << "[ERROR] Could not load Excel file for reset:" << filePath;
292 return;
293 }
294 QSqlDatabase db = QSqlDatabase::database();
295 if (!db.transaction()) {
296 qDebug() << "Transaction failed:" << db.lastError().text();
297 return;
298 }
299 QSqlQuery query;
300 // 1. Wipe the specific tables in order to respect potential constraints
301 QStringList tables = {"campusDistances", "souvenirs", "campusList"};
302 for (const QString &table : tables) {
303 if (!query.exec(QString("DELETE FROM %1").arg(table))) {
304 qDebug() << "Failed to wipe table" << table << ":" << query.lastError().text();
305 db.rollback();
306 return;
307 }
308 }
309 // 2. Iterate through Excel sheets and perform the direct insert
310 for (const QString &sheetName : xlsx.sheetNames()) {
311 xlsx.selectSheet(sheetName);
312 // Get Headers
313 QStringList columnNames;
314 QStringList quotedColumns;
315 QStringList placeholders;
316 int col = 1;
317 while (true) {
318 QVariant headerCell = xlsx.read(1, col);
319 if (!headerCell.isValid() || headerCell.isNull()) break;
320 QString colName = headerCell.toString();
321 columnNames << colName;
322 quotedColumns << QString("\"%1\"").arg(colName);
323 placeholders << "?";
324 col++;
325 }
326 if (columnNames.isEmpty()) continue;
327 // Prepare the INSERT query
328 QString sql = QString("INSERT INTO %1 (%2) VALUES (%3)")
329 .arg(sheetName,
330 quotedColumns.join(", "),
331 placeholders.join(", "));
332 query.prepare(sql);
333 // 3. Process Rows (Starting from Row 2)
334 int row = 2;
335 while (true) {
336 QVariant firstCell = xlsx.read(row, 1);
337 if (!firstCell.isValid() || firstCell.isNull()) break;
338 for (int c = 1; c <= columnNames.size(); ++c) {
339 query.addBindValue(xlsx.read(row, c));
340 }
341 if (!query.exec()) {
342 qDebug() << "[SQL ERROR] Reset Row" << row << "in" << sheetName << ":" << query.lastError().text();
343 }
344 row++;
345 }
346 }
347 if (db.commit()) {
348 qDebug() << "Database successfully reset and reloaded from Excel.";
349 } else {
350 qDebug() << "Commit failed during reset. Rolling back.";
351 db.rollback();
352 }
