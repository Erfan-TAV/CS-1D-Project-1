425void PlanPage::renderTrip() {
426 // 1. Get the layout from the scroll area
427 QHBoxLayout *layout = qobject_cast<QHBoxLayout*>(ui->scrollAreaWidgetContents->layout());
428 if (!layout) {
429 qDebug() << "[ERROR] Could not find layout for scrollAreaWidgetContents";
430 return;
431 }
432
433 // 2. Clear previous UI elements to prevent "ghosting" from old trips
434 QLayoutItem *item;
435 while ((item = layout->takeAt(0)) != nullptr) {
436 if (item->widget()) delete item->widget();
437 delete item;
438 }
439
440 // 3. Fetch the trip sequence from the temporary trip table
441 // We order by visitOrder to ensure the tour displays in the correct path sequence
442 QSqlQuery query("SELECT campusID, campusName, visitOrder FROM tripCampuses ORDER BY visitOrder ASC");
443
444 while (query.next()) {
445 int currentID = query.value("campusID").toInt();
446 QString name = query.value("campusName").toString();
447 int currentOrder = query.value("visitOrder").toInt();
448
449 int distanceToNext = 0;
450
451 // 4. Find the NEXT campus in the sequence to get the distance
452 QSqlQuery nextQuery;
453 nextQuery.prepare("SELECT campusID FROM tripCampuses WHERE visitOrder = :nextOrder");
454 nextQuery.bindValue(":nextOrder", currentOrder + 1);
455
456 if (nextQuery.exec() && nextQuery.next()) {
457 int nextID = nextQuery.value(0).toInt();
458
459 // REFACTORED: Use the helper function from databaseHelper.h
460 // This handles the bi-directional SQL check (ID1->ID2 or ID2->ID1)
461 double distResult = getDistanceBetween(currentID, nextID);
462
463 // Check for the "Not Found" sentinel value (999999.0)
464 if (distResult < 999998.0) {
465 distanceToNext = static_cast<int>(distResult);
466 } else {
467 qDebug() << "[WARNING] No distance found between ID" << currentID << "and" << nextID;
468 }
469 }
470
471 // 5. Add the campus stop widget
472 // The 'true' flag ensures an arrow and distance label are drawn after this stop
473 layout->addWidget(createStopWidget(name, distanceToNext, true));
474 }
475
476 // 6. Add the final "Goal" flag
477 // We pass 'false' for showArrow because no path follows the final destination
478 layout->addWidget(createStopWidget("🏁 Trip Finished", 0, false));
479
480 // 7. Add stretch at the end to keep the trip sequence left-aligned
481 layout->addStretch();
482
483 qDebug() << "[UI] Trip rendering complete.";
484}
